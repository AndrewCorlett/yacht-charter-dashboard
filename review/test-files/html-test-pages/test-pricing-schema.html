Manual Schema Deployment

If the schema fields are missing, run this SQL in your Supabase SQL Editor:

-- Add enhanced pricing fields to pricing_rules table
ALTER TABLE pricing_rules ADD COLUMN IF NOT EXISTS season_type TEXT CHECK (season_type IN ('high', 'low', 'other'));
ALTER TABLE pricing_rules ADD COLUMN IF NOT EXISTS season_start_date DATE;
ALTER TABLE pricing_rules ADD COLUMN IF NOT EXISTS season_end_date DATE;
ALTER TABLE pricing_rules ADD COLUMN IF NOT EXISTS charter_total DECIMAL(10,2);
ALTER TABLE pricing_rules ADD COLUMN IF NOT EXISTS deposit_amount DECIMAL(10,2);
ALTER TABLE pricing_rules ADD COLUMN IF NOT EXISTS security_deposit DECIMAL(10,2);
